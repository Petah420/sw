-- SAFE ANTI-KICK SETUP
--------------------------------------------------------------------------------
pcall(function()
    if hookmetamethod and getnamecallmethod then
        local oldKick
        oldKick = hookmetamethod(game, "__namecall", function(self, ...)
            local method = getnamecallmethod()
            if method == "Kick" or method == "kick" then
                if self == LocalPlayer then return nil end
            end
            return oldKick(self, ...)
        end)
    end
end)

        local function detach()
            if renderConnection then
                renderConnection:Disconnect()
                renderConnection = nil

            if localChar then
                local humanoid = localChar:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    humanoid:SetStateEnabled(Enum.HumanoidStateType.Freefall, true)
                    humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                    humanoid.Sit = false
                end
            end
            attachedTarget = nil
        end

        local function stopSpectating()
            spectatingTarget = nil
            pcall(function()
                RunService:UnbindFromRenderStep("StealthSpectate")
            end)
            local localChar = LocalPlayer.Character
            if localChar then
                local humanoid = localChar:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    workspace.CurrentCamera.CameraSubject = humanoid
                end
            end
        end

        local function spectatePlayer(targetPlayer)
            stopSpectating()
            if not targetPlayer or targetPlayer == LocalPlayer then return end
            spectatingTarget = targetPlayer

            RunService:BindToRenderStep("StealthSpectate", Enum.RenderPriority.Camera.Value + 1, function()
                if spectatingTarget and spectatingTarget.Character then
                    local targetHum = spectatingTarget.Character:FindFirstChildOfClass("Humanoid")
                    if targetHum then
                        workspace.CurrentCamera.CameraSubject = targetHum
                    end
                else
                    stopSpectating()
                end
            end)
        end

        local function attachToTarget(targetPlayer)
            detach()
            if not targetPlayer or targetPlayer == LocalPlayer then return end
            attachedTarget = targetPlayer

            renderConnection = RunService.RenderStepped:Connect(function(dt)
                local localChar = LocalPlayer.Character
                local targetChar = attachedTarget and attachedTarget.Character

                if not (localChar and targetChar) then
                    detach()
                    return
                end

                local localHRP = localChar:FindFirstChild("HumanoidRootPart")
                local localHumanoid = localChar:FindFirstChildOfClass("Humanoid")
                local targetHead = targetChar:FindFirstChild("Head")
                local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")

                if localHRP and targetHead and localHumanoid then
                    localHumanoid:SetStateEnabled(Enum.HumanoidStateType.Freefall, false)
                    if localHumanoid:GetState() == Enum.HumanoidStateType.Freefall then
                        localHumanoid:ChangeState(Enum.HumanoidStateType.Running)
                    end
                    
                    if attachMode ~= "Underground" then
                        localHumanoid.Sit = true
                    else
                        localHumanoid.Sit = false
                    end

                    for _, part in ipairs(localChar:GetChildren()) do
                        if part:IsA("BasePart") then
                            part.CanCollide = false
                        end
                    end

                    local offsetCFrame
                    if attachMode == "FaceSit" then
                        local dynamicZ = -0.3
                        if humping then
                            humpOffset = humpOffset + (dt * 6 * humpDirection)
                            if humpOffset > 0.4 then
                                humpOffset = 0.4
                                humpDirection = -1
                            elseif humpOffset < -0.3 then
                                humpOffset = -0.3
                                humpDirection = 1
                            end
                            dynamicZ = dynamicZ + humpOffset
                        end
                        offsetCFrame = CFrame.new(0, 0.5, dynamicZ) * CFrame.Angles(0, math.rad(180), 0)
                        localHRP.CFrame = targetHead.CFrame * offsetCFrame
                    elseif attachMode == "Shoulders" then
                        offsetCFrame = CFrame.new(0, 1.8, 0.2) * CFrame.Angles(0, 0, 0)
                        localHRP.CFrame = targetHead.CFrame * offsetCFrame
                    elseif attachMode == "Underground" then
                        if targetHRP then
                            local targetPos = targetHRP.Position
                            local rot = targetHRP.CFrame - targetPos
                            localHRP.CFrame = CFrame.new(targetPos.X, targetPos.Y - 12, targetPos.Z) * rot
                        end
                    end

                    localHRP.AssemblyLinearVelocity = Vector3.zero
                    localHRP.AssemblyAngularVelocity = Vector3.zero
                else
                    detach()
                end
            end)
        end

        local function playEmoteAnimation(animId)
            local char = LocalPlayer.Character

            if currentEmoteTrack then
                currentEmoteTrack:Stop()
                currentEmoteTrack = nil
            end

            local animator = humanoid:FindFirstChildOfClass("Animator") or Instance.new("Animator", humanoid)
            local anim = Instance.new("Animation")
            anim.AnimationId = "rbxassetid://" .. tostring(animId)

            local success, track = pcall(function()
                return animator:LoadAnimation(anim)
            end)

            if success and track then
                currentEmoteTrack = track
                track.Looped = true
                track:Play()
            end
        end

                local bg = hrp:FindFirstChild("FlyGyro")
                if bv and bg then
                    local camera = workspace.CurrentCamera
                    local moveDir = Vector3.zero

                    if UserInputService:IsKeyDown(Enum.KeyCode.W) then
                        moveDir = moveDir + camera.CFrame.LookVector
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.S) then
                        moveDir = moveDir - camera.CFrame.LookVector
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.A) then
                        moveDir = moveDir - camera.CFrame.RightVector
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.D) then
                        moveDir = moveDir + camera.CFrame.RightVector
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                        moveDir = moveDir + Vector3.new(0, 1, 0)
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
                        moveDir = moveDir - Vector3.new(0, 1, 0)
                    end

                    bv.Velocity = moveDir.Magnitude > 0 and (moveDir.Unit * flySpeedValue) or Vector3.zero
                    bg.CFrame = camera.CFrame
                end
            end
        end)

        task.spawn(function()
            local configSuccess, v1 = pcall(function()
                return require(ReplicatedStorage:WaitForChild("HalloweenEventConfig"))
            end)
            if not configSuccess or not v1 then
                warn("[HalloweenAuto DEBUG] CRITICAL: Failed to load HalloweenEventConfig from ReplicatedStorage!")
                return
            else
                print("[HalloweenAuto DEBUG] Config successfully loaded. WorldFolderName:", tostring(v1.WorldFolderName), "| RemotesFolderName:", tostring(v1.RemotesFolderName))
            end

            local remotes = ReplicatedStorage:WaitForChild(v1.RemotesFolderName)
            local trickAsk = remotes:WaitForChild("TrickAsk")
            local eventRemote = remotes:WaitForChild("Event")
            local getState = remotes:WaitForChild("GetState")
            local accept = remotes:WaitForChild("Accept")
            local turnIn = remotes:WaitForChild("TurnIn")

            eventRemote.OnClientEvent:Connect(function(action, data)
                if type(data) == "table" then
                    local responseAction = action or data.action
                    if (responseAction == "TrickAnswer" or responseAction == "TrickResult") or (type(data) == "table" and data.treat ~= nil) then
                        pendingResponse = data
                    end
                end
            end)

            local function teleportToPlayer(targetPlayer)
                local character = LocalPlayer.Character
                local targetCharacter = targetPlayer.Character
                if character and character:FindFirstChild("HumanoidRootPart") and targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart") then
                    character.HumanoidRootPart.CFrame = targetCharacter.HumanoidRootPart.CFrame + Vector3.new(0, 3, 0)
                    return true
                end
                return false
            end

            while task.wait(1.5) do
                local success, state = pcall(function()
                    return getState:InvokeServer()
                end)

                if not success or type(state) ~= "table" then
                    warn("[HalloweenAuto DEBUG] Failed to fetch state from server or state is not a table!")
                else
                    if autoAcceptEnabled and not state.accepted and not state.completed then
                        print("[HalloweenAuto DEBUG] Quest not accepted. Firing Accept remotes...")
                        local accSucc, accMsg = pcall(function() return accept:InvokeServer() end)
                        print("[HalloweenAuto DEBUG] Accept result:", accSucc, accMsg)
                        task.wait(1)
                    end

                    if state.accepted and not state.completed then
                        local litTable = type(state.lit) == "table" and state.lit or {}
                        local treatedTable = type(state.treated) == "table" and state.treated or {}
                        local litCount = #litTable
                        local treatCount = #treatedTable

                        statusLabel.Text = string.format("Lanterns: %d/%d | Treats: %d/%d (No Fish)", litCount, v1.LanternsNeeded, treatCount, v1.TreatsNeeded)

                        -- LANTERN DEBUG SCANNING
                        if autoLanternsEnabled and litCount < v1.LanternsNeeded then
                            local worldFolder = workspace:FindFirstChild(v1.WorldFolderName)
                            if not worldFolder then
                                warn("[HalloweenAuto DEBUG] Workspace folder not found by name:", tostring(v1.WorldFolderName))
                            else
                                local lanternsFolder = worldFolder and worldFolder:FindFirstChild("Lanterns")
                                if not lanternsFolder then
                                    warn("[HalloweenAuto DEBUG] 'Lanterns' subfolder not found inside workspace folder:", tostring(v1.WorldFolderName))
                                else
                                    local character = LocalPlayer.Character
                                    local rootPart = character and character:FindFirstChild("HumanoidRootPart")
                                    
                                    for _, lantern in ipairs(lanternsFolder:GetChildren()) do
                                        if lantern:IsA("Model") then
                                            local prompt = lantern:FindFirstChild("LightPrompt", true)
                                            local lightPart = lantern:FindFirstChild("Light")
                                            
                                            if prompt and prompt:IsA("ProximityPrompt") then
                                                local lanternId = lantern:GetAttribute("LanternId") or lantern.Name
                                                local alreadyLit = false
                                                
                                                for _, id in ipairs(litTable) do
                                                    if id == lanternId then 
                                                        alreadyLit = true 
                                                        break 
                                                    end
                                                end

                                                if not alreadyLit then
                                                    if lightPart and lightPart:IsA("BasePart") and rootPart then
                                                        print("[HalloweenAuto DEBUG] Lighting unlit lantern:", lantern.Name, "| ID:", tostring(lanternId))
                                                        local oldCFrame = rootPart.CFrame
                                                        rootPart.CFrame = lightPart.CFrame + Vector3.new(0, 3, 0)
                                                        task.wait(0.2)
                                                        if fireproximityprompt then
                                                            fireproximityprompt(prompt)
                                                        else
                                                            prompt:InputHoldBegin()
                                                            task.wait(prompt.HoldDuration + 0.1)
                                                            prompt:InputHoldEnd()
                                                        end
                                                        task.wait(0.3)
                                                        rootPart.CFrame = oldCFrame
                                                        task.wait(0.2)
                                                    else
                                                        warn("[HalloweenAuto DEBUG] Lantern found ('" .. lantern.Name .. "'), but missing 'Light' part or RootPart!")
                                                    end
                                                end
                                            end
                                        end
                                    end
                                end
                            end
                        end

                        if autoTreatsEnabled and treatCount < v1.TreatsNeeded then
                            for _, player in ipairs(Players:GetPlayers()) do
                                if player ~= LocalPlayer then
                                    if teleportToPlayer(player) then
                                        task.wait(0.3)
                                    end
                                    pcall(function() trickAsk:FireServer(player.UserId) end)
                                    pendingResponse = nil
                                    local timeout = 5
                                    local elapsed = 0
                                    while elapsed < timeout do
                                        task.wait(0.5)
                                        elapsed = elapsed + 0.5
                                        if pendingResponse then
                                            local resultStr = pendingResponse.result or pendingResponse.choice or pendingResponse.type or pendingResponse[1]
                                            if resultStr == "treat" or pendingResponse.treat == true then
                                                treatsReceivedCount = treatsReceivedCount + 1
                                            end
                                            break
                                        end
                                    end
                                    break
                                end
                            end
                        end

                        if autoTurnInEnabled and litCount >= v1.LanternsNeeded and treatCount >= v1.TreatsNeeded then
                            print("[HalloweenAuto DEBUG] Requirements met! Firing TurnIn...")
                            local turnSucc, turnMsg = pcall(function() return turnIn:InvokeServer() end)
                            print("[HalloweenAuto DEBUG] TurnIn result:", turnSucc, turnMsg)
                        end
                    end
                end
            end
        end)

        print("Universal Hub UI Loaded Successfully!")
    end)

    if not success then
        warn("UI Load Error: " .. tostring(err))
    end
